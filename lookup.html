6.4 Student Search Using a Binary Search Tree

The key is compared with the root and the search moves to the left or right subtree. SEARCH(root, key) 1. If root == NULL, return NOT_FOUND 2. If key == root->id, return root 3. If key < root->id return SEARCH(root->left, key) Else return SEARCH(root->right, key)

6.7 Hashing Using the Division Method and Linear Probing
A colliding key is placed in the next free slot of the table. HASH(key) = key % TABLE_SIZE                      (TABLE_SIZE = 101) INSERT(key) 1. index = HASH(key) 2. While table[index] is occupied index = (index + 1) % TABLE_SIZE 3. table[index] = key SEARCH(key) 1. index = HASH(key) 2. While table[index] is occupied If table[index] == key, return index index = (index + 1) % TABLE_SIZE 3. Return NOT_FOUND

7.5 Student List, Search Tree, Hash Table and Graph Helpers
/* Linear search in the linked list. steps returns the comparisons made. */
static Student *find_student(int id, int *steps)
{
    Student *t = head;
    int count = 0;

    while (t != NULL) {
        count++;
        if (t->id == id)
            break;
        t = t->next;
    }
    if (steps != NULL)
        *steps = count;
    return t;
}

static const char *student_name(int id)
{
    Student *s = find_student(id, NULL);
    return s ? s->name : "Unknown";
}

/* ---- binary search tree ---- */

static TreeNode *bst_insert(TreeNode *r, Student *s)
{
    if (r == NULL) {
        TreeNode *n = (TreeNode *)safe_malloc(sizeof(TreeNode));
        n->id = s->id;
        n->rec = s;
        n->left = n->right = NULL;
        return n;
    }





    if (s->id < r->id)
         r->left = bst_insert(r->left, s);
    else
         r->right = bst_insert(r->right, s);
    return r;
}

static Student *bst_search(int id, int *steps)
{
    TreeNode *t = root;
    int count = 0;

    while (t != NULL) {
        count++;
        if (id == t->id) {
            *steps = count;
            return t->rec;
        }
        t = (id < t->id) ? t->left : t->right;
    }
    *steps = count;
    return NULL;
}

static void bst_inorder(TreeNode *r)
{
    if (r == NULL)
        return;
    bst_inorder(r->left);
    printf(" %-12d %s\n", r->id, r->rec->name);
    bst_inorder(r->right);
}

static void bst_free(TreeNode *r)
{
    if (r == NULL)
        return;
    bst_free(r->left);
    bst_free(r->right);
    free(r);
}

/* ---- hash table (division method, linear probing) ---- */

static int hash_insert(int id)
{
    int index = id % TABLE_SIZE;
    int probes = 1;

    while (hash_table[index] != 0) {
        index = (index + 1) % TABLE_SIZE;
        probes++;
    }
    hash_table[index] = id;
    return probes;
}

static int hash_search(int id, int *probes)
{
    int index = id % TABLE_SIZE;
    int count = 1;

    while (hash_table[index] != 0) {
        if (hash_table[index] == id) {
            *probes = count;
            return index;
        }
        index = (index + 1) % TABLE_SIZE;
        count++;





        if (count > TABLE_SIZE)
            break;
    }
    *probes = count;
    return -1;
}

/* ---- graph helpers ---- */

static int graph_index(int id)
{
    int i;
    for (i = 0; i < vcount; i++)
        if (graph_ids[i] == id)
            return i;
    return -1;
}
