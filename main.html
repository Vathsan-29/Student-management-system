7.16 Memory Clean-up and Main Function

static void free_all(void)
{
    Student *s;
    Message *m;
    Chat *c;
    UndoNode *u;
    Member *g, *start;

    while (head != NULL) { s = head; head = head->next; free(s); }
    bst_free(root);
    while (q_front != NULL) { m = q_front; q_front = q_front->next; free(m); }
    while (chat_head != NULL) { c = chat_head; chat_head = chat_head->next; free(c); }
    while (undo_top != NULL) { u = undo_top; undo_top = undo_top->next; free(u); }
    if (group_last != NULL) {
        start = group_last->next;
        group_last->next = NULL;            /* break the circle */
        while (start != NULL) { g = start; start = start->next; free(g); }
    }
}

int main(void)
{
    int choice;

    do {
        show_menu();
        choice = read_int("Enter your choice: ");
        switch (choice) {
            case 1: register_student();        break;
            case 2: display_students();        break;
            case 3: search_student();          break;
            case 4: sort_students();           break;
            case 5: send_message();            break;
            case 6: view_incoming();           break;
            case 7: view_chat_history();       break;
            case 8: undo_message();            break;
            case 9: manage_group();            break;
            case 10: manage_connections();     break;
            case 11: find_connection_path();   break;
            case 12: manage_notifications();   break;
            case 13: hash_menu();              break;
            case 14: stack_demo();             break;
            case 0: printf("Thank you for using the system.\n"); break;
            default: printf("Invalid choice. Please try again.\n");
        }
    } while (choice != 0);

    free_all();
    return 0;
}
