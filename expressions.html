Array stack and recursion

Data structureStack (array), recursion
Syllabus unitUnit II

An array stack converts an infix expression to postfix and evaluates it, with checks for wrong expressions and division by zero. Recursion solves the Tower of Hanoi.

Algorithm

6.8 Infix to Postfix Conversion

A stack holds the operators until their turn to be written to the output. TO_POSTFIX(infix) 1. Create an empty stack 2. For each character c of the expression If c is a digit : add c to the output If c is '(' : push c If c is ')' : pop to the output until '(' is found If c is + - * / : while the top has equal or higher precedence, pop it to the output; then push c 3. Pop all remaining operators to the output

7.15 Stack and Expression Demonstration (Option 14)
static int precedence(char op)
{
    if (op == '+' || op == '-')
        return 1;
    if (op == '*' || op == '/')
        return 2;
    return 0;
}

static int to_postfix(const char *infix, char *postfix)
{





    char stack[EXPR_MAX];
    int top = -1, k = 0, i;
    char c;

    for (i = 0; infix[i] != '\0'; i++) {
        c = infix[i];
        if (c == ' ')
            continue;
        if (isdigit((unsigned char)c)) {
            postfix[k++] = c;
        } else if (c == '(') {
            stack[++top] = c;
        } else if (c == ')') {
            while (top >= 0 && stack[top] != '(')
                 postfix[k++] = stack[top--];
            if (top < 0)
                 return 0;                    /* missing '(' */
            top--;                            /* discard '(' */
        } else if (c == '+' || c == '-' || c == '*' || c == '/') {
            while (top >= 0 && precedence(stack[top]) >= precedence(c))
                 postfix[k++] = stack[top--];
            stack[++top] = c;
        } else {
            return 0;                         /* illegal character */
        }
    }
    while (top >= 0) {
        if (stack[top] == '(')
            return 0;                         /* missing ')' */
        postfix[k++] = stack[top--];
    }
    postfix[k] = '\0';
    return 1;
}

static int eval_postfix(const char *postfix, int *result)
{
    int stack[EXPR_MAX], top = -1, i, a, b;
    char c;

    for (i = 0; postfix[i] != '\0'; i++) {
        c = postfix[i];
        if (isdigit((unsigned char)c)) {
            stack[++top] = c - '0';
        } else {
            if (top < 1)
                 return 0;                   /* not enough operands */
            b = stack[top--];
            a = stack[top--];
            if (c == '+')
                 stack[++top] = a + b;
            else if (c == '-')
                 stack[++top] = a - b;
            else if (c == '*')
                 stack[++top] = a * b;
            else {
                 if (b == 0)
                     return -1;              /* division by zero */
                 stack[++top] = a / b;
            }
        }
    }
    if (top != 0)
        return 0;
    *result = stack[top];
    return 1;
}

static void hanoi(int n, char from, char to, char via)





{
    if (n == 0)
        return;
    hanoi(n - 1, from, via, to);
    printf(" Move disk %d from %c to %c\n", n, from, to);
    hanoi(n - 1, via, to, from);
}

static void stack_demo(void)
{
    int choice, n, result, status;
    char infix[EXPR_MAX], postfix[EXPR_MAX];

    printf("\n 1. Infix to postfix, then evaluate\n");
    printf(" 2. Tower of Hanoi\n");
    choice = read_int("Enter your choice: ");

    if (choice == 1) {
        read_line("Enter infix expression (single-digit numbers, + - * / and brackets): ",
                   infix, sizeof infix);
        if (!to_postfix(infix, postfix)) {
             printf("Invalid expression.\n");
             return;
        }
        printf("Postfix expression : %s\n", postfix);
        status = eval_postfix(postfix, &result);
        if (status == 1)
             printf("Value               : %d\n", result);
        else if (status == -1)
             printf("Division by zero.\n");
        else
             printf("Invalid expression.\n");
    } else if (choice == 2) {
        n = read_int("Number of disks (1-6): ");
        if (n < 1 || n > 6) {
             printf("Please enter a value from 1 to 6.\n");
             return;
        }
        hanoi(n, 'A', 'C', 'B');
        printf("Total moves: %d\n", (1 << n) - 1);
    } else {
        printf("Invalid choice.\n");
    }
}
