Messaging, chat history and undo

Queue, doubly linked list and stack

Data structureQueue, doubly linked list, stack
Syllabus unitUnit II
Menu option5, 6, 7, 8

Every sent message is stored in three structures: the queue for delivery in arrival order, the doubly linked chat history that can be read in both directions, and the undo stack that takes back the latest message.

Algorithm

6.2 Message Queue (FIFO)

Messages wait in a linked queue and leave in the same order in which they arrived. ENQUEUE(message) 1. Create a message node, new->next = NULL 2. If rear == NULL front = rear = new Else rear->next = new rear = new DEQUEUE() 1. If front == NULL, report "queue empty" and stop 2. temp = front 3. front = front->next 4. If front == NULL, rear = NULL 5. Deliver the message and free temp

6.3 Undo Using a Stack (LIFO)
The ID of every sent message is pushed on a stack, so the latest message is the first one to be undone. PUSH(message_id) 1. Create a stack node 2. node->next = top 3. top = node POP() 1. If top == NULL, report "nothing to undo" and stop 2. temp = top, top = top->next 3. Remove the message temp->msg_id from the queue (if present) 4. Remove the message temp->msg_id from the chat history 5. Free temp

static void enqueue_message(Message *m)
{
    m->next = NULL;
    if (q_rear == NULL)
        q_front = q_rear = m;
    else {





        q_rear->next = m;
        q_rear = m;
    }
}

static void send_message(void)
{
    int from, to;
    Message *m;
    Chat *c;
    UndoNode *u;
    char text[200];

    if (student_count < 2) {
        printf("At least two registered students are needed.\n");
        return;
    }
    from = read_int("Sender ID   : ");
    if (find_student(from, NULL) == NULL) {
        printf("Sender is not registered.\n");
        return;
    }
    to = read_int("Receiver ID : ");
    if (find_student(to, NULL) == NULL) {
        printf("Receiver is not registered.\n");
        return;
    }
    if (from == to) {
        printf("You cannot send a message to yourself.\n");
        return;
    }
    read_line("Message     : ", text, sizeof text);
    if (text[0] == '\0') {
        printf("Message cannot be empty.\n");
        return;
    }

    /* 1. put the message in the queue */
    m = (Message *)safe_malloc(sizeof(Message));
    m->msg_id = next_msg_id;
    m->sender = from;
    m->receiver = to;
    strcpy(m->text, text);
    enqueue_message(m);

    /* 2. add it to the end of the chat history */
    c = (Chat *)safe_malloc(sizeof(Chat));
    c->msg_id = next_msg_id;
    c->sender = from;
    c->receiver = to;
    strcpy(c->text, text);
    c->next = NULL;
    c->prev = chat_tail;
    if (chat_tail == NULL)
         chat_head = c;
    else
         chat_tail->next = c;
    chat_tail = c;

    /* 3. push it on the undo stack */
    u = (UndoNode *)safe_malloc(sizeof(UndoNode));
    u->msg_id = next_msg_id;
    u->next = undo_top;
    undo_top = u;

    printf("Message %d sent from %s to %s.\n", next_msg_id,
           student_name(from), student_name(to));
    next_msg_id++;
}





static void view_incoming(void)
{
    Message *t, *m;
    int choice, n = 0;

    if (q_front == NULL) {
        printf("No pending messages.\n");
        return;
    }
    printf("\n 1. Show pending messages\n");
    printf(" 2. Deliver the oldest message\n");
    choice = read_int("Enter your choice: ");

    if (choice == 1) {
        for (t = q_front; t != NULL; t = t->next) {
             n++;
             printf(" %d. [Msg %d] %s -> %s : %s\n", n, t->msg_id,
                    student_name(t->sender), student_name(t->receiver), t->text);
        }
        printf("Pending messages: %d\n", n);
    } else if (choice == 2) {
        m = q_front;                         /* dequeue */
        q_front = q_front->next;
        if (q_front == NULL)
             q_rear = NULL;
        printf("Delivered to %s: \"%s\" (from %s)\n",
                student_name(m->receiver), m->text, student_name(m->sender));
        free(m);
    } else {
        printf("Invalid choice.\n");
    }
}

static void view_chat_history(void)
{
    Chat *t;
    int a, choice, shown = 0;

    if (chat_head == NULL) {
        printf("No messages in the chat history.\n");
        return;
    }
    a = read_int("Enter a student ID to see only their chats (0 for all): ");
    printf(" 1. Oldest first\n 2. Newest first\n");
    choice = read_int("Enter your choice: ");
    if (choice != 1 && choice != 2) {
        printf("Invalid choice.\n");
        return;
    }

    t = (choice == 1) ? chat_head : chat_tail;
    while (t != NULL) {
        if (a == 0 || t->sender == a || t->receiver == a) {
            printf(" [Msg %d] %s -> %s : %s\n", t->msg_id,
                    student_name(t->sender), student_name(t->receiver), t->text);
            shown++;
        }
        t = (choice == 1) ? t->next : t->prev;
    }
    if (shown == 0)
        printf("No messages found for this student.\n");
}

static void undo_message(void)
{
    UndoNode *u;
    Message *m, *prev = NULL;
    Chat *c;





    int id, in_queue = 0;

    if (undo_top == NULL) {
        printf("Nothing to undo.\n");
        return;
    }
    u = undo_top;                           /* pop */
    undo_top = u->next;
    id = u->msg_id;
    free(u);

    /* remove from the queue if it is still waiting there */
    for (m = q_front; m != NULL; prev = m, m = m->next) {
        if (m->msg_id == id) {
            if (prev == NULL)
                 q_front = m->next;
            else
                 prev->next = m->next;
            if (m == q_rear)
                 q_rear = prev;
            free(m);
            in_queue = 1;
            break;
        }
    }

    /* remove from the chat history */
    for (c = chat_tail; c != NULL; c = c->prev) {
        if (c->msg_id == id) {
            if (c->prev == NULL)
                 chat_head = c->next;
            else
                 c->prev->next = c->next;
            if (c->next == NULL)
                 chat_tail = c->prev;
            else
                 c->next->prev = c->prev;
            free(c);
            break;
        }
    }

    if (in_queue)
         printf("Message %d was undone and removed from the pending queue.\n", id);
    else
         printf("Message %d was undone (it was already delivered).\n", id);
}
