Connections and shortest path

Graph with BFS and DFS

Data structureGraph (adjacency matrix)
Syllabus unitUnit IV
Menu option10, 11

Students are vertices and connections are edges in an adjacency matrix. BFS and DFS traverse the network, and BFS finds the shortest connection path using a parent array.

Algorithm

6.5 Shortest Connection Path Using BFS

Breadth-first search reaches students level by level, so the first path found to the target is the shortest. FIND_PATH(a, b) 1. Mark all students unvisited, parent[all] = -1 2. Mark a visited and add a to the queue 3. While queue is not empty and b is not visited v = remove from queue For each unvisited student i connected to v Mark i visited, parent[i] = v Add i to the queue 4. If b is not visited, print "no path" and stop 5. Walk from b back to a using parent[] and print the path

static void dfs_visit(int v, int visited[])
{
    int i;
    visited[v] = 1;
    printf("%s ", student_name(graph_ids[v]));
    for (i = 0; i < vcount; i++)
        if (adj[v][i] && !visited[i])
            dfs_visit(i, visited);
}

static void manage_connections(void)
{
    int choice, a, b, i, j, found;
    int visited[MAX_STUDENTS] = {0};

    if (vcount < 2) {
        printf("At least two registered students are needed.\n");
        return;
    }
    printf("\n 1. Add a connection\n 2. Display all connections\n");
    printf(" 3. BFS traversal from a student\n");
    printf(" 4. DFS traversal from a student\n");
    choice = read_int("Enter your choice: ");

    if (choice == 1) {
        a = graph_index(read_int("First student ID : "));
        b = graph_index(read_int("Second student ID : "));
        if (a < 0 || b < 0)
            printf("Both students must be registered.\n");
        else if (a == b)
            printf("A student cannot connect to themselves.\n");
        else if (adj[a][b])
            printf("These students are already connected.\n");
        else {
            adj[a][b] = adj[b][a] = 1;
            printf("%s and %s are now connected.\n",
                   student_name(graph_ids[a]), student_name(graph_ids[b]));
        }
    } else if (choice == 2) {
        for (i = 0; i < vcount; i++) {
            printf(" %s :", student_name(graph_ids[i]));
            found = 0;
            for (j = 0; j < vcount; j++)
                if (adj[i][j]) {
                    printf(" %s,", student_name(graph_ids[j]));
                    found = 1;
                }
            if (!found)
                printf(" (no connections)");
            printf("\n");
        }
    } else if (choice == 3) {
        int queue[MAX_STUDENTS], front = 0, rear = 0, v;

        a = graph_index(read_int("Start student ID: "));
        if (a < 0) {
            printf("Student is not registered.\n");
            return;
        }
        visited[a] = 1;





        queue[rear++] = a;
        printf("BFS order: ");
        while (front < rear) {
             v = queue[front++];
             printf("%s ", student_name(graph_ids[v]));
             for (i = 0; i < vcount; i++)
                 if (adj[v][i] && !visited[i]) {
                     visited[i] = 1;
                     queue[rear++] = i;
                 }
        }
        printf("\n");
    } else if (choice == 4) {
        a = graph_index(read_int("Start student ID: "));
        if (a < 0) {
             printf("Student is not registered.\n");
             return;
        }
        printf("DFS order: ");
        dfs_visit(a, visited);
        printf("\n");
    } else {
        printf("Invalid choice.\n");
    }
}

static void find_connection_path(void)
{
    int a, b, i, v, front = 0, rear = 0;
    int queue[MAX_STUDENTS], parent[MAX_STUDENTS], visited[MAX_STUDENTS] = {0};
    int path[MAX_STUDENTS], len = 0;

    if (vcount < 2) {
        printf("At least two registered students are needed.\n");
        return;
    }
    a = graph_index(read_int("From student ID: "));
    b = graph_index(read_int("To student ID : "));
    if (a < 0 || b < 0) {
        printf("Both students must be registered.\n");
        return;
    }

    for (i = 0; i < vcount; i++)
        parent[i] = -1;
    visited[a] = 1;
    queue[rear++] = a;
    while (front < rear && !visited[b]) {       /* BFS gives shortest path */
        v = queue[front++];
        for (i = 0; i < vcount; i++)
            if (adj[v][i] && !visited[i]) {
                visited[i] = 1;
                parent[i] = v;
                queue[rear++] = i;
            }
    }

    if (!visited[b]) {
        printf("No connection path exists between these students.\n");
        return;
    }
    for (v = b; v != -1; v = parent[v])         /* walk back to the start */
        path[len++] = v;

    printf("Shortest path (%d step(s)): ", len - 1);
    for (i = len - 1; i >= 0; i--) {
        printf("%s", student_name(graph_ids[path[i]]));
        if (i > 0)
            printf(" -> ");





     }
     printf("\n");
 }
