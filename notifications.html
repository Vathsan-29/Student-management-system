Algorithm

6.6 Priority Notifications Using a Max Heap

The heap always keeps the highest priority at the top, so the most urgent notification is given out first. INSERT_HEAP(value) 1. Place value at the end of the heap, i = last index 2. While i > 0 and heap[parent(i)] < heap[i] Swap heap[parent(i)] and heap[i] i = parent(i) EXTRACT_MAX() 1. top = heap[0] 2. Move the last element to heap[0] and reduce the size by 1 3. i = 0 4. While heap[i] has a child child = the larger of the two children If heap[i] >= heap[child], stop Swap heap[i] and heap[child], i = child 5. Return top

 static void heap_insert(int priority, const char *text)
 {
     int i = heap_size++;
     Notice tmp;

     heap[i].priority = priority;
     strcpy(heap[i].text, text);
     while (i > 0 && heap[(i - 1) / 2].priority < heap[i].priority) {
         tmp = heap[i];
         heap[i] = heap[(i - 1) / 2];
         heap[(i - 1) / 2] = tmp;
         i = (i - 1) / 2;
     }
 }

 static Notice heap_extract_max(void)
 {
     Notice top = heap[0], tmp;
     int i = 0, child;

     heap[0] = heap[--heap_size];
     while (1) {
         child = 2 * i + 1;
         if (child >= heap_size)
             break;
         if (child + 1 < heap_size && heap[child + 1].priority > heap[child].priority)
             child++;
         if (heap[i].priority >= heap[child].priority)
             break;
         tmp = heap[i];
         heap[i] = heap[child];
         heap[child] = tmp;
         i = child;
     }
     return top;
 }

 static void manage_notifications(void)
 {
     int choice, p, i;
     char text[100];
     Notice n;

     printf("\n 1. Add a notification\n");
     printf(" 2. Read the most urgent notification\n");
     printf(" 3. Show all notifications (heap order)\n");
     choice = read_int("Enter your choice: ");

     if (choice == 1) {
         if (heap_size >= HEAP_MAX) {
             printf("Notification box is full.\n");
             return;
         }
         read_line("Notification text: ", text, sizeof text);
         if (text[0] == '\0') {
             printf("Text cannot be empty.\n");
             return;
         }
         p = read_int("Priority (1 = low, 10 = very urgent): ");
         if (p < 1 || p > 10) {
             printf("Priority must be between 1 and 10.\n");
             return;
         }





        heap_insert(p, text);
        printf("Notification added.\n");
    } else if (choice == 2) {
        if (heap_size == 0) {
             printf("No notifications.\n");
             return;
        }
        n = heap_extract_max();
        printf("Most urgent (priority %d): %s\n", n.priority, n.text);
    } else if (choice == 3) {
        if (heap_size == 0) {
             printf("No notifications.\n");
             return;
        }
        for (i = 0; i < heap_size; i++)
             printf(" [%d] priority %d : %s\n", i, heap[i].priority, heap[i].text);
    } else {
        printf("Invalid choice.\n");
    }
}
