Algorithm

6.1 Student Registration (Singly Linked List)

A new student is added at the end of the list. The same record is then inserted into the search tree, the hash table and the graph. INSERT_STUDENT(head, data) 1. Create new node 2. If memory is not available, report error and stop 3. Copy data into the new node 4. new->next = NULL 5. If head == NULL head = new Else temp = head While temp->next != NULL temp = temp->next temp->next = new 6. Return head

7.6 Student Registration (Option 1)
static void register_student(void)
{
    Student *n, *t;
    int id, year, i, j;

    if (student_count >= MAX_STUDENTS) {
        printf("Student limit (%d) reached.\n", MAX_STUDENTS);
        return;
    }

    id = read_int("Enter student ID (positive number): ");
    if (id <= 0) {
        printf("ID must be greater than zero.\n");
        return;
    }
    if (find_student(id, NULL) != NULL) {
        printf("A student with this ID already exists.\n");
        return;
    }

    n = (Student *)safe_malloc(sizeof(Student));
    n->id = id;
    read_line("Enter name           : ", n->name, sizeof n->name);
    read_line("Enter department     : ", n->dept, sizeof n->dept);
    if (n->name[0] == '\0' || n->dept[0] == '\0') {
        printf("Name and department cannot be empty.\n");
        free(n);
        return;
    }
    year = read_int("Enter year (1-4)     : ");
    if (year < 1 || year > 4) {
        printf("Year must be between 1 and 4.\n");
        free(n);
        return;
    }
    n->year = year;
    n->next = NULL;

    /* insert at the end of the singly linked list */
    if (head == NULL) {
        head = n;
    } else {
        t = head;
        while (t->next != NULL)
             t = t->next;
        t->next = n;
    }
    student_count++;





    /* keep the other structures in step with the list */
    root = bst_insert(root, n);
    hash_insert(id);
    i = vcount++;
    graph_ids[i] = id;
    for (j = 0; j < MAX_STUDENTS; j++)
        adj[i][j] = adj[j][i] = 0;

    printf("Student registered successfully.\n");
}
7.7 Display Student Details (Option 2)
static void display_students(void)
{
    Student *t = head;

    if (t == NULL) {
        printf("No students registered yet.\n");
        return;
    }
    line('-', 70);
    printf("%-12s %-20s %-24s %-4s\n", "ID", "NAME", "DEPARTMENT", "YEAR");
    line('-', 70);
    while (t != NULL) {
        printf("%-12d %-20s %-24s %-4d\n", t->id, t->name, t->dept, t->year);
        t = t->next;
    }
    line('-', 70);
    printf("Total students: %d\n", student_count);
}
