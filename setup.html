The file starts with the libraries and constants, then defines one structure for every data structure: Student for the list, TreeNode for the search tree, Message for the queue, Chat for the history, UndoNode for the stack, Member for the group and Notice for the heap. Global pointers and arrays keep the state of the program.

Source code

7.1 Program Header, Libraries and Constants

/*
 * STUDENT COMMUNICATION SYSTEM
 * IT23302 - Data Structures Using C
 * Department of Information Technology, MIT Campus, Anna University
 *
 * A menu-driven console program that uses the data structures from the
 * syllabus to run a small student messaging system.
 *
 *   Singly linked list   - student records
 *   Binary search tree   - student search by ID
 *   Queue (linked)       - pending (incoming) messages
 *   Doubly linked list   - chat history
 *   Stack (linked)       - undo of sent messages
 *   Circular linked list - study group, speaker rotation
 *   Graph (adj. matrix) - student connections, BFS, DFS, shortest path
 *   Max heap             - priority notifications
 *   Hash table           - quick lookup, linear probing
 *   Stack (array)        - infix to postfix and postfix evaluation
 *   Recursion            - Tower of Hanoi
 *
 * Compile : gcc student_communication_system.c -o scs
 * Run     : ./scs
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <limits.h>

#define MAX_STUDENTS 50
#define TABLE_SIZE   101
#define HEAP_MAX     50
#define EXPR_MAX     100

7.2 Structures

 typedef struct Student {              /* node of the student list */
     int id;
     char name[50];
     char dept[30];
     int year;
     struct Student *next;
 } Student;

 typedef struct TreeNode {             /* node of the binary search tree */
     int id;
     Student *rec;
     struct TreeNode *left, *right;
 } TreeNode;

 typedef struct Message {              /* node of the message queue */
     int msg_id, sender, receiver;
     char text[200];
     struct Message *next;
 } Message;

 typedef struct Chat {                 /* node of the chat history list */
     int msg_id, sender, receiver;
     char text[200];





    struct Chat *prev, *next;
} Chat;

typedef struct UndoNode {              /* node of the undo stack */
    int msg_id;
    struct UndoNode *next;
} UndoNode;

typedef struct Member {                /* node of the group (circular list) */
    int id;
    struct Member *next;
} Member;

typedef struct Notice {                /* one entry of the notification heap */
    int priority;
    char text[100];
} Notice;
