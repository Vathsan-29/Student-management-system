static int in_group(int id)
{
    Member *p;
    if (group_last == NULL)
         return 0;
    p = group_last->next;
    do {
         if (p->id == id)
             return 1;
         p = p->next;
    } while (p != group_last->next);
    return 0;
}

static void manage_group(void)
{
    int choice, id;
    Member *n, *p, *prev;





   printf("\n 1. Add a member\n 2. Remove a member\n");
   printf(" 3. Show group members\n 4. Pass turn to next speaker\n");
   choice = read_int("Enter your choice: ");

   if (choice == 1) {
       id = read_int("Enter student ID to add: ");
       if (find_student(id, NULL) == NULL) {
            printf("Student is not registered.\n");
       } else if (in_group(id)) {
            printf("Student is already in the group.\n");
       } else {
            n = (Member *)safe_malloc(sizeof(Member));
            n->id = id;
            if (group_last == NULL) {
                n->next = n;
                group_last = group_cur = n;
            } else {
                n->next = group_last->next;
                group_last->next = n;
                group_last = n;
            }
            printf("%s added to the group.\n", student_name(id));
       }
   } else if (choice == 2) {
       if (group_last == NULL) {
            printf("The group is empty.\n");
            return;
       }
       id = read_int("Enter student ID to remove: ");
       prev = group_last;
       p = group_last->next;
       do {
            if (p->id == id)
                break;
            prev = p;
            p = p->next;
       } while (prev != group_last);

       if (p->id != id) {
            printf("Student is not in the group.\n");
            return;
       }
       if (p->next == p) {                   /* only one member */
            group_last = group_cur = NULL;
       } else {
            prev->next = p->next;
            if (p == group_last)
                group_last = prev;
            if (p == group_cur)
                group_cur = p->next;
       }
       free(p);
       printf("Member removed from the group.\n");
   } else if (choice == 3) {
       if (group_last == NULL) {
            printf("The group is empty.\n");
            return;
       }
       p = group_last->next;
       do {
            printf(" %d %s%s\n", p->id, student_name(p->id),
                    (p == group_cur) ? "   <-- current speaker" : "");
            p = p->next;
       } while (p != group_last->next);
   } else if (choice == 4) {
       if (group_cur == NULL) {
            printf("The group is empty.\n");
            return;
       }





        group_cur = group_cur->next;
        printf("It is now %s's turn to speak.\n", student_name(group_cur->id));
    } else {
        printf("Invalid choice.\n");
    }
}
